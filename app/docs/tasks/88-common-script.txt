# Shared setup of the task 88 runs (sourced by the other 88-*-script.txt; run from app/ after
# `cargo build --release --locked -p rom-app -p rom-cli`). Loopback only. Every window ends
# itself (--screenshot, --bench, ROM_NET_LOBBY_SECONDS, ROM_NET_QUIT_AFTER); window pictures
# are taken with ImageMagick `import -window <id>` (id from `xwininfo -root -tree`); nothing is
# sent to any window. Each process gets its own config and data folder.
T=${T:-$PWD/target/t88-run}          # state, keys and logs of the runs
B=target/release/rise-of-man
R=target/release/rom
run() {  # run <name> <env...> -- <args...>: one capped app process in its own folders
  local name=$1; shift; local envs=()
  while [ "$1" != "--" ]; do envs+=("$1"); shift; done; shift
  mkdir -p "$T/$name"
  env -u WAYLAND_DISPLAY XDG_CONFIG_HOME="$T/$name" XDG_DATA_HOME="$T/$name" "${envs[@]}" \
    systemd-run --user --scope -q -p MemoryMax=6G -p MemorySwapMax=0 "$B" "$@" \
    > "$T/$name.log" 2>&1
}
window() { xwininfo -root -tree | grep -m1 -E "$1" | awk '{print $1}'; }
shot() {  # shot <title pattern> <file>: a picture of our own window, never a click to pick one
  local id; id=$(window "$1"); [ -n "$id" ] && import -window "$id" "$2" || echo "no window $1"
}
wait_file() { for _ in $(seq 1 120); do [ -s "$1" ] && return 0; sleep 0.5; done; return 1; }
